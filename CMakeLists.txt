cmake_minimum_required(VERSION 3.24)
project(batbox VERSION 0.1.2 LANGUAGES C CXX)
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 20)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
  set(CMAKE_OBJCXX_EXTENSIONS OFF)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(BUILD_TESTING "Build indigenous workflow checks" ON)
if(WIN32)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  # One section per function and object, so the release link can drop the unused ones.
  add_compile_options($<$<NOT:$<CONFIG:Debug>>:-ffunction-sections>
    $<$<NOT:$<CONFIG:Debug>>:-fdata-sections>)
endif()
include(cmake/WindowsPkgConfig.cmake)
find_package(nlohmann_json 3.11 REQUIRED)
find_package(Threads REQUIRED)
find_package(CURL REQUIRED)
include(cmake/Libraries.cmake)
include(cmake/IndigenousApplication.cmake)
include(cmake/Install.cmake)
if(BUILD_TESTING)
  include(cmake/Tests.cmake)
endif()
